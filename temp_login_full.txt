"use client";

import { useEffect, useState } from "react";
import Link from "next/link";
import { LazyMotion, domAnimation, m } from "framer-motion";
import { useRouter, useSearchParams } from "next/navigation";
import { zodResolver } from "@hookform/resolvers/zod";
import { useForm } from "react-hook-form";
import { z } from "zod";
import {
  ArrowLeft,
  ArrowRight,
  BarChart3,
  CheckCircle2,
  Eye,
  EyeOff,
  Loader2,
  LockKeyhole,
  Mail,
  ShieldCheck,
  TrendingUp,
  Users,
} from "lucide-react";
import PublicAudiencePageFrame from "@/components/public/PublicAudiencePageFrame";
import { Input } from "@/components/ui/input";
import { useAuth } from "@/context/AuthContext";
import { usePublicPageData } from "@/hooks/usePublicPageData";
import { useResolvedPublicAudience } from "@/hooks/useResolvedPublicAudience";
import { getApiErrorMessage, isApiConnectivityError } from "@/lib/api-errors";
import { withPublicAudience } from "@/lib/public-audience";
import { canonicalizeProtectedPath, getDefaultAppPath } from "@/lib/erp/route-config";
import { getDeviceId } from "@/services/apiClient";
import { authApi } from "@/services/modules";

const loginSchema = z.object({
  email: z.string().email("Enter a valid work email"),
  password: z.string().min(6, "Password must be at least 6 characters"),
});

type LoginValues = z.infer<typeof loginSchema>;

type LoginContent = {
  title: string;
  subtitle: string;
  loginButtonLabel: string;
};

const leftFeatures = [
  { icon: Users, text: "HR, Payroll & Attendance in one place", desc: "Automate timesheets and salary payouts." },
  { icon: BarChart3, text: "Finance, inventory & procurement unified", desc: "Real-time cost tracking and audit trails." },
  { icon: ShieldCheck, text: "Role-based access with audit trails", desc: "Enterprise governance and workflow controls." },
  { icon: TrendingUp, text: "Real-time dashboards and analytics", desc: "One-click operational reporting." },
];

export default function LoginPage() {
  const router = useRouter();
  const searchParams = useSearchParams();
  const { refreshUser } = useAuth();
  const nextPath = searchParams.get("next");
  const { audience } = useResolvedPublicAudience();
  const { profile, headerContent, footerContent, allowedPaths, pageVisible, loading, parseContent } =
    usePublicPageData("login", audience);
  const content = parseContent<LoginContent>({
    title: "Welcome back",
    subtitle: "Sign in to your workspace to continue.",
    loginButtonLabel: "Sign in",
  });

  const [errorMessage, setErrorMessage] = useState("");
  const [showPassword, setShowPassword] = useState(false);
  const [csrfReady, setCsrfReady] = useState(false);

  const emailFieldId = "login-email";
  const passwordFieldId = "login-password";

  const {
    register,
    handleSubmit,
    formState: { errors, isSubmitting },
  } = useForm<LoginValues>({
    resolver: zodResolver(loginSchema),
    defaultValues: { email: "", password: "" },
  });

  useEffect(() => {
    let cancelled = false;
    authApi.csrfToken().finally(() => {
      if (!cancelled) setCsReady();
    });
    return () => { cancelled = true; };
  }, []);

  const setCsReady = () => {
    setCsrfReady(true);
  };

  const onSubmit = async (values: LoginValues) => {
    setErrorMessage("");
    try {
      await authApi.login({
        email: values.email.trim(),
        password: values.password,
        deviceId: getDeviceId(),
      });
      await refreshUser();
      router.push(nextPath ? canonicalizeProtectedPath(nextPath) : getDefaultAppPath());
    } catch (error: unknown) {
      if (isApiConnectivityError(error)) {
        setErrorMessage("Unable to reach the server. Please check your connection and try again.");
        return;
      }
      setErrorMessage(getApiErrorMessage(error, "Invalid email or password. Please try again."));
    }
  };

  const brandName = profile.brandName || profile.companyName || "UabIndia ERP";
  const homeHref = withPublicAudience("/", audience);

  return (
    <PublicAudiencePageFrame
      audience={audience}
      profile={profile}
      headerContent={headerContent}
      footerContent={footerContent}
      allowedPaths={allowedPaths}
      pageVisible={pageVisible}
      showHeader={false}
      showFooter={false}
    >
      <div className="grid min-h-screen grid-cols-1 lg:grid-cols-2 bg-slate-50">
        {/* ── Left Brand & Info Panel (Desktop) ─────────────────────────── */}
        <div className="relative hidden overflow-hidden lg:flex lg:h-full flex-col justify-between border-r border-slate-200 bg-gradient-to-b from-indigo-50/70 via-slate-50/60 to-white px-12 py-8 xl:py-12 w-full">
          {/* Ambient mesh glow blobs */}
          <div className="absolute -left-16 -bottom-16 h-64 w-64 rounded-full bg-cyan-500/8 blur-[70px] pointer-events-none" />
          <div className="absolute -right-16 -top-16 h-64 w-64 rounded-full bg-indigo-500/10 blur-[70px] pointer-events-none" />

          {/* Floating Widget 1: SSL Secure (Top Right) */}
          <LazyMotion features={domAnimation}>
            <m.div
              animate={{ y: [0, -8, 0] }}
              transition={{ repeat: Infinity, duration: 4.8, ease: "easeInOut" }}
              className="absolute top-12 right-12 hidden xl:flex items-center gap-3.5 rounded-2xl border border-slate-200 bg-white p-3.5 shadow-[0_12px_30px_rgba(15,23,42,0.06)] z-20"
            >
              <div className="flex h-8.5 w-8.5 items-center justify-center rounded-xl bg-emerald-50 border border-emerald-100 text-emerald-600">
                <ShieldCheck className="h-4.5 w-4.5" />
              </div>
              <div className="text-left">
                <p className="text-[9px] font-bold text-slate-400 uppercase tracking-wider">SSL Secure</p>
                <p className="text-xs font-bold text-slate-800">Session Encrypted</p>
              </div>
            </m.div>
          </LazyMotion>

          {/* Floating Widget 2: Employees Online (Bottom Right) */}
          <LazyMotion features={domAnimation}>
            <m.div
              animate={{ y: [0, 8, 0] }}
              transition={{ repeat: Infinity, duration: 4.5, ease: "easeInOut", delay: 1 }}
              className="absolute bottom-12 right-12 hidden xl:flex items-center gap-3.5 rounded-2xl border border-slate-200 bg-white p-3.5 shadow-[0_12px_30px_rgba(15,23,42,0.06)] z-20"
            >
              <div className="flex h-8.5 w-8.5 items-center justify-center rounded-xl bg-indigo-50 border border-indigo-100 text-indigo-600">
                <Users className="h-4.5 w-4.5" />
              </div>
              <div className="text-left">
                <p className="text-[9px] font-bold text-slate-400 uppercase tracking-wider">Employees Online</p>
                <p className="text-xs font-bold text-slate-800">128 Online Now</p>
              </div>
            </m.div>
          </LazyMotion>

          {/* Floating Widget 3: HR Payroll Attendance (Lower Left) */}
          <LazyMotion features={domAnimation}>
            <m.div
              animate={{ y: [0, -6, 0] }}
              transition={{ repeat: Infinity, duration: 5.2, ease: "easeInOut", delay: 0.5 }}
              className="absolute bottom-28 left-8 hidden xl:flex items-center gap-3.5 rounded-2xl border border-slate-200 bg-white p-3.5 shadow-[0_12px_30px_rgba(15,23,42,0.06)] z-20"
            >
              <div className="flex h-8.5 w-8.5 items-center justify-center rounded-xl bg-cyan-50 border border-cyan-100 text-cyan-600">
                <BarChart3 className="h-4.5 w-4.5" />
              </div>
              <div className="text-left">
                <p className="text-[9px] font-bold text-slate-400 uppercase tracking-wider">HR & Payroll</p>
                <p className="text-xs font-bold text-slate-800">Unified Operations</p>
              </div>
            </m.div>
          </LazyMotion>

          <div className="relative flex flex-col h-full justify-between z-10">
            {/* Top Bar: Logo & Navigation */}
            <div className="flex items-center justify-between w-full">
              <Link href={homeHref} className="flex items-center gap-2 group w-fit">
                <img
                  src="/logo.png"
                  alt={brandName}
                  className="h-6 w-auto object-contain transition group-hover:opacity-85"
                />
              </Link>
              <Link
                href={homeHref}
                className="flex items-center gap-1.5 rounded-lg border border-slate-200 bg-white px-3.5 py-1.5 text-xs font-semibold text-slate-600 transition-all duration-300 hover:border-indigo-300 hover:text-indigo-600 hover:bg-slate-50"
              >
                <ArrowLeft className="h-3.5 w-3.5" />
                Back to site
              </Link>
            </div>

            {/* Core Product Features Block */}
            <div className="my-auto py-8 space-y-7 max-w-[520px] w-full">
              <div className="space-y-3.5">
                <div className="inline-flex items-center gap-2 rounded-full border border-indigo-100 bg-indigo-50 px-3 py-1 text-[0.65rem] font-bold uppercase tracking-[0.2em] text-indigo-700 shadow-sm">
                  <span className="h-1.5 w-1.5 animate-pulse rounded-full bg-indigo-600" />
                  {audience === "tenant" ? "Your workspace" : "Platform Control"}
                </div>
                <h1 className="text-[2.25rem] font-extrabold leading-[1.1] tracking-tight text-slate-900">
                  {audience === "tenant" ? (
                    <>
                      Your business,<br />
                      <span className="bg-gradient-to-r from-indigo-600 to-indigo-700 bg-clip-text text-transparent">
                        one platform.
                      </span>
                    </>
                  ) : (
                    <>
                      Platform<br />
                      <span className="bg-gradient-to-r from-indigo-600 to-indigo-700 bg-clip-text text-transparent">
                        control center.
                      </span>
                    </>
                  )}
                </h1>
                <p className="text-[0.92rem] leading-6 text-slate-500 max-w-[420px] w-full">
                  {audience === "tenant"
                    ? "HR, payroll, attendance, finance, and workflows — all unified under a clean operating system."
                    : "Manage instances, customer accounts, and security baselines from a single operational console."}
                </p>
              </div>

              {/* Premium feature rows */}
              <div className="space-y-3">
                {leftFeatures.map(({ icon: Icon, text, desc }) => (
                  <div 
                    key={text} 
                    className="group flex items-start gap-3.5 rounded-xl border border-slate-200/50 bg-white/70 p-3 shadow-sm transition-all duration-200 hover:bg-white hover:border-indigo-100 hover:shadow-indigo-500/5"
                  >
                    <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-indigo-50 border border-indigo-100 text-indigo-600">
                      <Icon className="h-4.5 w-4.5" />
                    </div>
                    <div>
                      <h3 className="text-xs font-bold text-slate-800 transition-colors">{text}</h3>
                      <p className="mt-0.5 text-[10px] text-slate-400 leading-none">{desc}</p>
                    </div>
                  </div>
                ))}
              </div>
            </div>

            {/* Stats section */}
            <div className="grid grid-cols-3 gap-3">
              {[
                { value: "500+", label: "Clients" },
                { value: "12+", label: "Modules" },
                { value: "99.9%", label: "Uptime" },
              ].map((stat) => (
                <div key={stat.label} className="rounded-xl border border-slate-200/50 bg-white/70 px-3 py-3.5 text-center shadow-sm hover:border-indigo-100 transition-all">
                  <p className="text-xl font-extrabold text-indigo-600 leading-none">{stat.value}</p>
                  <p className="mt-1 text-[9px] font-bold uppercase tracking-wider text-slate-400">{stat.label}</p>
                </div>
              ))}
            </div>
          </div>
        </div>

        {/* ── Right Panel: Modern Login Card ───────────────────────────── */}
        <div className="flex flex-1 flex-col bg-slate-50 lg:h-full lg:overflow-y-auto">
          {/* Mobile Navigation Header */}
          <div className="flex items-center justify-between border-b border-slate-200 bg-white px-5 py-3.5 lg:hidden">
            <Link href={homeHref} className="flex items-center gap-2">
              <img src="/logo.png" alt={brandName} className="h-5 w-auto object-contain" />
            </Link>
            <Link
              href={homeHref}
              className="flex items-center gap-1.5 rounded-lg border border-slate-200 bg-white px-3 py-1.5 text-xs font-semibold text-slate-600 transition hover:border-indigo-300 hover:text-indigo-600"
            >
              <ArrowLeft className="h-3.5 w-3.5" />
              Back
            </Link>
          </div>

          {/* Centralized Card Form Area */}
          <div className="flex flex-1 flex-col items-center justify-center px-5 md:px-10 lg:px-16 py-10">
            <div className="w-full max-w-[520px]">
              {/* Login Card */}
              <div className="overflow-hidden rounded-[28px] border border-slate-200 bg-white p-12 shadow-[0_20px_50px_rgba(15,23,42,0.08),0_1px_4px_rgba(15,23,42,0.01)] relative">
                {/* Top color accent border */}
                <div className="absolute top-0 inset-x-0 h-1.5 bg-gradient-to-r from-indigo-500 via-indigo-600 to-indigo-700" />

                {/* Form Header */}
                <div className="mb-6">
                  <div className="mb-3 inline-flex items-center gap-2 rounded-full border border-indigo-100 bg-indigo-50 px-3.5 py-1.5 shadow-sm">
                    <div className="h-1.5 w-1.5 animate-pulse rounded-full bg-indigo-600" />
                    <span className="text-[0.65rem] font-bold uppercase tracking-[0.2em] text-indigo-700">
                      Secure access
                    </span>
                  </div>
                  <h2 className="text-[1.7rem] font-extrabold tracking-tight text-slate-900">{content.title}</h2>
                  <p className="mt-1.5 text-sm text-slate-500 leading-relaxed">{content.subtitle}</p>
                </div>

                {/* Login Form */}
                <form className="space-y-6" onSubmit={handleSubmit(onSubmit)} noValidate>
                  {/* Email Input Field */}
                  <div className="space-y-3">
                    <label className="block text-xs font-bold uppercase tracking-wider text-slate-600" htmlFor={emailFieldId}>
                      Email address
                    </label>
                    <div className="relative group">
                      <Mail className="absolute left-4 top-1/2 h-5 w-5 -translate-y-1/2 text-slate-400 group-focus-within:text-indigo-600 transition-colors" />
                      <Input
                        id={emailFieldId}
                        {...register("email")}
                        className="h-14 w-full rounded-2xl border-slate-200 bg-slate-50/50 pl-12 pr-4 text-sm text-slate-900 shadow-none placeholder:text-slate-400 transition-all focus:bg-white focus:border-indigo-500 focus:ring-4 focus:ring-indigo-100/50"
                        placeholder="name@company.com"
                        type="email"
                        autoComplete="email"
                        autoFocus
                      />
                    </div>
                    {errors.email && (
                      <p className="mt-1 text-xs font-semibold text-rose-600">{errors.email.message}</p>
                    )}
                  </div>

                  {/* Password Input Field */}
                  <div className="space-y-3">
                    <div className="flex items-center justify-between">
                      <label className="block text-xs font-bold uppercase tracking-wider text-slate-600" htmlFor={passwordFieldId}>
                        Password
                      </label>
                      <Link
                        href={withPublicAudience("/forgot-password", audience)}
                        className="text-xs font-bold text-indigo-600 hover:text-indigo-700 transition-colors"
                      >
                        Forgot password?
                      </Link>
                    </div>
                    <div className="relative group">
                      <LockKeyhole className="absolute left-4 top-1/2 h-5 w-5 -translate-y-1/2 text-slate-400 group-focus-within:text-indigo-600 transition-colors" />
                      <Input
                        id={passwordFieldId}
                        {...register("password")}
                        className="h-14 w-full rounded-2xl border-slate-200 bg-slate-50/50 pl-12 pr-12 text-sm text-slate-900 shadow-none placeholder:text-slate-400 transition-all focus:bg-white focus:border-indigo-500 focus:ring-4 focus:ring-indigo-100/50"
                        type={showPassword ? "text" : "password"}
                        placeholder="••••••••"
                        autoComplete="current-password"
                      />
                      <button
                        type="button"
                        onClick={() => setShowPassword((v) => !v)}
                        className="absolute right-4 top-1/2 -translate-y-1/2 rounded-lg p-1.5 text-slate-400 hover:text-slate-700 transition-colors"
                        aria-label={showPassword ? "Hide password" : "Show password"}
                        tabIndex={-1}
                      >
                        {showPassword ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
                      </button>
                    </div>
                    {errors.password && (
                      <p className="mt-1 text-xs font-semibold text-rose-600">{errors.password.message}</p>
                    )}
                  </div>

                  {/* Error Notification */}
                  {errorMessage && (
                    <div className="rounded-xl border border-rose-100 bg-rose-50 px-4 py-3 text-xs font-semibold text-rose-700 leading-relaxed shadow-sm">
                      {errorMessage}
                    </div>
                  )}

                  {/* Submit Button */}
                  <button
                    type="submit"
                    disabled={isSubmitting || !csrfReady}
                    className="mt-6 flex h-14 w-full items-center justify-center gap-2 rounded-2xl bg-gradient-to-r from-indigo-600 to-indigo-700 text-sm font-bold text-white shadow-[0_4px_14px_rgba(79,70,229,0.35)] transition-all duration-200 hover:from-indigo-700 hover:to-indigo-800 hover:shadow-[0_6px_20px_rgba(79,70,229,0.45)] disabled:opacity-50 disabled:cursor-not-allowed hover:-translate-y-0.5 active:translate-y-0 active:scale-[0.98] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-indigo-500 focus-visible:ring-offset-2"
                  >
                    {isSubmitting ? (
                      <span className="flex items-center justify-center gap-2">
                        <Loader2 className="h-4 w-4 animate-spin" />
                        Signing in...
                      </span>
                    ) : !csrfReady ? (
                      <span className="flex items-center justify-center gap-2">
                        <Loader2 className="h-4 w-4 animate-spin" />
                        Preparing...
                      </span>
                    ) : (
                      <span className="flex items-center justify-center gap-2">
                        {content.loginButtonLabel}
                        <ArrowRight className="h-4 w-4" />
                      </span>
                    )}
                  </button>
                </form>

                {/* Card footer / Registration redirect */}
                <div className="mt-8 space-y-3.5">
                  <div className="flex items-center gap-3">
                    <div className="h-px flex-1 bg-slate-200" />
                    <span className="text-[10px] font-bold uppercase tracking-wider text-slate-400">or</span>
                    <div className="h-px flex-1 bg-slate-200" />
                  </div>

                  <p className="text-center text-sm text-slate-500">
                    Need access?{" "}
                    <Link
                      href={withPublicAudience("/register", audience)}
                      className="font-bold text-indigo-600 hover:text-indigo-700 transition-colors"
                    >
                      Request workspace
                    </Link>
                  </p>
                </div>
              </div>

              {/* Security Shield Info */}
              <div className="mt-5 flex items-center justify-center gap-2">
                <CheckCircle2 className="h-4 w-4 text-emerald-500" />
                <span className="text-xs font-semibold text-slate-400">Secured with enterprise-grade SSL encryption</span>
              </div>

              {/* Desktop-only back link (under card) */}
              <div className="mt-5 hidden lg:flex justify-center">
                <Link
                  href={homeHref}
                  className="flex items-center gap-2 text-sm font-semibold text-slate-400 hover:text-indigo-600 transition-all hover:gap-2.5"
                >
                  <ArrowLeft className="h-4 w-4" />
                  Back to {brandName} website
                </Link>
            </div>
          </div>
        </div>
      </div>
    </div>
  </PublicAudiencePageFrame>
  );
}
